cmake_minimum_required(VERSION 3.20)
project(PDFMerger VERSION 1.2.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This application supports Windows only.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(merge_core src/merger.cpp)
target_include_directories(merge_core PUBLIC src)
target_compile_definitions(merge_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(merge_core PUBLIC gdiplus ole32)
add_executable(OrderedPdfMerger WIN32 src/main.cpp)
configure_file(static/app-icon.ico ${CMAKE_CURRENT_BINARY_DIR}/app-icon.ico COPYONLY)
configure_file(src/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
target_sources(OrderedPdfMerger PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
target_link_libraries(OrderedPdfMerger PRIVATE merge_core comctl32 comdlg32 shell32 user32 gdi32 uuid)
if(MSVC)
  target_compile_options(merge_core PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(OrderedPdfMerger PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(merge_core PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  target_compile_options(OrderedPdfMerger PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  target_link_options(OrderedPdfMerger PRIVATE -municode -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(merger_tests tests/merger_tests.cpp)
  target_link_libraries(merger_tests PRIVATE merge_core)
  if(MSVC)
    target_compile_options(merger_tests PRIVATE /W4 /utf-8)
  else()
    target_link_options(merger_tests PRIVATE -municode -static)
  endif()
  add_test(NAME merger_tests COMMAND merger_tests)
endif()
